cmake_minimum_required(VERSION 3.16)
project(credits C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CREDITS_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
enable_testing()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-Wall -Wextra -Wpedantic -fstack-usage -ffp-contract=off)
endif()
if(CREDITS_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(credits_core src/memory.c src/random.c src/canvas.c src/scheduler.c src/data.c src/credits.c src/ocean.c src/text.c src/scenes.c src/weather.c src/player.c)
target_include_directories(credits_core PUBLIC src)
add_executable(credits src/main.c src/terminal.c src/host_time.c src/host_console.c)
target_link_libraries(credits PRIVATE credits_core)
add_executable(semantics_probe tests/probe.c src/terminal.c)
target_link_libraries(semantics_probe PRIVATE credits_core)
add_executable(scheduler_probe tests/scheduler_probe.c)
target_link_libraries(scheduler_probe PRIVATE credits_core)
add_executable(player_probe tests/player_probe.c)
target_link_libraries(player_probe PRIVATE credits_core)
add_executable(scene_probe tests/scene_probe.c)
target_link_libraries(scene_probe PRIVATE credits_core)
if(WIN32)
  add_executable(console_test tests/console_test.c)
endif()
add_test(NAME smoke COMMAND credits --self-test)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME ansi_oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/ansi_test.py)
add_test(NAME canvas_random COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/semantics.py $<TARGET_FILE:semantics_probe>)
add_test(NAME scheduler COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/scheduler_test.py $<TARGET_FILE:scheduler_probe>)
add_test(NAME exported_data COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/export_data.py --check)
add_test(NAME exported_events COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/export_events.py --check)
add_test(NAME initialized_data COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/data_test.py $<TARGET_FILE:credits> 1)
add_test(NAME ocean_text COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/compare.py $<TARGET_FILE:credits> --last 1079 --seeds 0,1,42)
add_test(NAME terminal_ansi COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/ansi_replay_test.py $<TARGET_FILE:credits>)
add_test(NAME full_timeline COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/compare.py $<TARGET_FILE:credits> --last 6508 --seeds 0,1,42 --jumps 1,2,3,4,5,6)
add_test(NAME all_scenes COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/scene_test.py $<TARGET_FILE:scene_probe>)
foreach(jump RANGE 1 6)
  add_test(NAME player_${jump} COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/player_test.py $<TARGET_FILE:player_probe> --seed 1 --jump ${jump})
endforeach()
if(WIN32)
  add_test(NAME windows_console COMMAND console_test $<TARGET_FILE:credits> ${CMAKE_BINARY_DIR}/console-finish.ansi ${CMAKE_BINARY_DIR}/console-stop.ansi)
endif()
