project(credits C)

option(CREDITS_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

option(CREDITS_SDL "Build SDL2 display and audio host" OFF)
option(CREDITS_STANDALONE "Windows EXE with static SDL and embedded WAV" OFF)
set(CREDITS_WAV "${CMAKE_SOURCE_DIR}/credits.wav" CACHE FILEPATH "WAV embedded in standalone build")
if(CREDITS_SDL)
  set(SDL2_NO_MWINDOWS ON)
  find_package(SDL2 CONFIG REQUIRED)
  add_executable(credits_sdl src/sdl_frontend.c)
  if(CREDITS_STANDALONE)
    if(NOT WIN32 OR NOT EXISTS "${CREDITS_WAV}")
      message(FATAL_ERROR "Standalone build requires Windows and an existing CREDITS_WAV")
    endif()
    enable_language(RC)
    configure_file(src/embedded_audio.rc.in embedded_audio.rc @ONLY)
    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/embedded_audio.rc" PROPERTIES OBJECT_DEPENDS "${CREDITS_WAV}")
    target_sources(credits_sdl PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/embedded_audio.rc")
    target_compile_definitions(credits_sdl PRIVATE CREDITS_EMBEDDED_AUDIO=1)
    target_link_libraries(credits_sdl PRIVATE credits_core60 SDL2::SDL2-static)
    if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
      target_link_options(credits_sdl PRIVATE -static)
    endif()
    add_test(NAME sdl_standalone COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/standalone_test.py $<TARGET_FILE:credits_sdl> "${CREDITS_WAV}")
  else()
    target_link_libraries(credits_sdl PRIVATE credits_core60 SDL2::SDL2)
  endif()
  add_test(NAME sdl_audio_transport COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/audio_test.py $<TARGET_FILE:credits_sdl>)
  if(WIN32 AND NOT CREDITS_STANDALONE)
    add_custom_command(TARGET credits_sdl POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:credits_sdl>)
  endif()
endif()
